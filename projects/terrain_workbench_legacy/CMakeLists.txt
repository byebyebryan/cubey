add_executable(cubey_project_terrain_workbench_legacy
    main.cpp
    terrain_debug_export.cpp
    terrain_engine_reference.cpp
    terrain_generator.cpp
    terrain_phase_profile.cpp
    terrain_process_fields.cpp
    terrain_product.cpp
)
set_target_properties(cubey_project_terrain_workbench_legacy
    PROPERTIES OUTPUT_NAME terrain_workbench_legacy
)
target_include_directories(
    cubey_project_terrain_workbench_legacy
    PRIVATE
        "${CMAKE_CURRENT_SOURCE_DIR}"
)
target_link_libraries(
    cubey_project_terrain_workbench_legacy
    PRIVATE
        cubey::core
        cubey::engine
        nlohmann_json::nlohmann_json
        cubey_project_warnings
)

cubey_add_app_target(
    cubey_project_terrain_workbench_preview_legacy
    OUTPUT_NAME terrain_workbench_preview_legacy
    SHADER_DEFINE CUBEY_TERRAIN_WORKBENCH_PREVIEW_SHADER_DIR
    WINDOWED_SMOKE_TEST terrain_workbench_preview_legacy_reaches_glfw
    WINDOWED_SMOKE_PATTERN "terrain_workbench_preview_legacy: .*rendering terrain workbench preview"
    WINDOWED_SMOKE_ARGS
        --grid-size
        65
    SOURCES
        terrain_preview_main.cpp
        terrain_preview_app.cpp
        terrain_preview_config.cpp
        terrain_engine_reference.cpp
        terrain_preview_mesh.cpp
        terrain_generator.cpp
        terrain_phase_profile.cpp
        terrain_process_fields.cpp
        terrain_product.cpp
    SHADERS
        shaders/terrain_preview.vert
        shaders/terrain_preview.frag
        shaders/terrain_engine_runtime.vert
        shaders/terrain_engine_runtime.frag
    SHADER_INCLUDE_DIRS
        "${CMAKE_CURRENT_SOURCE_DIR}/shaders"
    SHADER_DEPENDS
        "${CMAKE_CURRENT_SOURCE_DIR}/shaders/terrain_engine_reference.glsl"
)
target_link_libraries(
    cubey_project_terrain_workbench_preview_legacy
    PRIVATE
        nlohmann_json::nlohmann_json
)
